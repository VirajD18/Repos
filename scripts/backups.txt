------------------- DB Backup -------------------

(-U if user not postgres)

--- Directory ---

nohup pg_dump -h IP -p 6435 -d DATABASE -Fd -f /PATH/DB_backup -v > /tmp/DB_backup.log 2>&1 &

--- .SQL ---

nohup pg_dump -h IP -p 6435 -d DATABASE -f /PATH/DB_backup/DB_backup.sql -v > /tmp/DB_backup.log 2>&1 &

--- Schema Only ---

nohup pg_dump -h IP -p 6435 -d DATABASE -s -f /PATH/Schema_backup/Schema_backup.sql -v > /tmp/Schema_backup.log 2>&1 &

--- Perticular Schema Only ---

nohup pg_dump -h IP -p 6434 -d DATABASE -n Schema -s -f /PATH/Schema_backup/Schema_backup.sql -v > /tmp/Schema_backup.log 2>&1 &

--- Roles Only ---

nohup pg_dumpall -h IP -p 6434 -r -f /PATH/DB_roles/DB_roles_backup.sql -v > /tmp/DB_roles_backup.log 2>&1 &

--- Tables Only ---

nohup pg_dump -h IP -p 5432 -d DATABASE -t TABLE -f /PATH/Table_backup/Table_backup.sql -v > /tmp/Table_backup.log 2>&1 &

nohup pg_dump -h IP -p 5432 -d DATABASE -U USERNAME -t "schema.\"table\"" -f /PATH/Table_backup/Table_backup.sql -v > /tmp/Table_backup.log 2>&1 &
