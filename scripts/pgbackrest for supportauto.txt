11) Setting up daily maintenance, daily incr backup and weekly full backup scripts using crontab

[postgres@bflprspautodb01 bkp_logs]$ crontab -l
00 05 * * * sh /home/postgres/supp_maintenance.sh


#incr backup pgbackrest#

00 04 * * 1-6 sh /home/postgres/incr_bkp_support.sh

#full backup pgbackrest#

00 04 * * 0 sh /home/postgres/full_bkp_support.sh

------------------maintenance------------------

[postgres@bflprspautodb01 bkp_logs]$ cat /home/postgres/supp_maintenance.sh
#!/bin/bash

psql -p 6434 -d postgres -U postgres -e -a -c '\i /home/postgres/maintenance_script/supp_vacuum.sql' > /home/postgres/maintenance_script/supp_vacuum_log.log

[postgres@bflprspautodb01 ~]$ cat /home/postgres/maintenance_script/supp_vacuum.sql
set maintenance_work_mem='2GB';
vacuum verbose analyze;
reindex database CONCURRENTLY postgres;
reindex system postgres;
\c supportautomate
set maintenance_work_mem='2GB';
vacuum verbose analyze;
reindex database CONCURRENTLY supportautomate;
reindex system supportautomate;

------------------incr backup------------------

[postgres@bflprspautodb01 ~]$ cat /home/postgres/incr_bkp_support.sh
a=$(date +%F)

pgbackrest --stanza=supportautomate1 --log-level-console=info --type=incr backup > /backup/bkp_logs/incr_backup_support_$a.log

------------------full backup------------------

a=$(date +%F)
