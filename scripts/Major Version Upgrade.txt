Take one full backup before activity

2)intialize data directory to postgresql-16    ------/data/pgdata16

	/data/pgdata14
	mkdir /data/pgdata16
	chown postgres:postgres pgdata16
	chmod 700 pgdata16

	/usr/pgsql-16/bin/initdb -D /data/pgdata16/ -W
	
	Utilit@ri@n!4
	
3)change port of current prod postgresql-14 to 6435 and keep new cluster version 16 port to 6436.
 
4)make the changes in shared_preload_libraries of postgresql-16 

	shared_preload_libraries = 'repmgr,pg_stat_statements,credcheck,pgaudit'
 
 
5) check credcheck, pgaudit parametrss

	postgresql.auto.conf
	
	credcheck.password_min_length = '10'
	credcheck.password_min_special = '2'
	credcheck.password_min_digit = '2'
	credcheck.password_min_upper = '2'
	credcheck.password_min_lower = '2'
	credcheck.password_min_repeat = '1'
	credcheck.password_contain_username = 'on'
	log_rotation_size = '50MB'
	log_checkpoints = 'on'

	pgaudit.log = 'role'
	
6) Stop the antivirus
	
	systemctl stop mdatp.service
	
	mdatp exclusion file add --path /data/pgdata16
	mdatp exclusion file add --path /usr/pgsql-16

6) stop both the cluster.
 
----------------------------PERFORM UPGRADE----------------------------------------------

For 8 core -j 4
 
7)/usr/pgsql-16/bin/pg_upgrade -j 4 --old-datadir /data/pgdata14/ --new-datadir /data/pgdata16/ --user postgres --old-bindir  /usr/pgsql-14/bin/ --new-bindir /usr/pgsql-16/bin/ --old-port 6435 --new-port 6436 --check
 
8)nohup /usr/pgsql-16/bin/pg_upgrade -j 4 -v --old-datadir /data/pgdata14/ --new-datadir /data/pgdata16/ --user postgres --old-bindir  /usr/pgsql-14/bin/ --new-bindir /usr/pgsql-16/bin/ --old-port 6435 --new-port 6436 > /tmp/pgupgradeSLAPIDATA.log &
 
9)start the new cluster postgresql-16
 
	 /usr/pgsql-16/bin/pg_ctl -D /data/pgdata16/ start

10)run the update_extensions.sql which will be created by pg_upgrade utility.
	 
11) PERFORM vacuume full.
 
	Set :-
	
	maintenance_work_mem = 2GB

	SET maintenance_work_mem = '2GB';

	/usr/pgsql-16/bin/vacuumdb -U postgres --all --analyze-in-stages -j 4

		shared_buffers = 16GB

13) Setup Pg_hba file on prod 

    copy from old cluster to new
 
14) Set archive mode and archive_command
 
15) Create new stanza for pgbackrest

	pgbackrest --stanza=salesonedata2 stanza-create

	pgbackrest --stanza=salesonedata2 check
	
	*Current*
	
	[salesonedata1]
	pg1-path=/data/pgdata14
	pg1-port=6434

	*New*
	
	[salesonedata2]
	pg1-path=/data/pgdata16
	pg1-port=6434
 
16) Setup DR database server

	i.nohup /usr/pgsql-16/bin/pg_basebackup -v -P -D /data/pgdata16/ -h 10.156.32.22 -X stream -p 6434 -c fast -U postgres -R > /tmp/slapidatadrbuild.log &
	
    ii. start the server using 16 data directory
	
	iii. copy pg_hba.conf from old DR cluster to new cluster.
	
	
17)setup REPMGR

	i. copy repmgr.conf from /etc/repmgr/14/repmgr.conf to /etc/repmgr/16/repmgr.conf
        and make the neccessary changes and reregester primary and standby.
	
	ii. Register both the servers
	
		repmgr primary register
		
		repmgr standby register
		
18)setup bash profile for primary and standby.

# .bash_profile

# Get the aliases and functions
if [ -f ~/.bashrc ]; then
        . ~/.bashrc
fi
PATH=$PATH:$HOME/.local/bin:$HOME/bin

export PGDATA=/data/pgdata16
export PGDATABASE=postgres
export PGPORT=6434

# User specific environment and startup programs
umask 077
 
19) after the complete the activity take full backup using pgbackrest.
